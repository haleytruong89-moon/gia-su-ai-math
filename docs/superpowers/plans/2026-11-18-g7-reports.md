```sql
-- ADDITIVE and idempotent. Nothing is dropped except a CHECK constraint that is re-created wider.
-- 1. learning_cards.report_path: the tutor now has a parents' evening report page.
-- 2. teaching_rules.source_key / source_ids: where a proposed rule came from (no double proposals).
-- 3. usage_monthly.console_alert_sent_at: "Console out of money" push at most once a day (D23).
-- 4. model_calls.purpose += 'reference' (reference solution after self_solved).
-- 5. tutor.add_usage: atomic monthly spend (replaces read-then-write in the tutor app). service_role only.

-- 1 (same columns, same order as 0021: only report_path changes)
create or replace view public.learning_cards with (security_invoker = true) as
select
  family_id, child_id, date_vn as local_date, 'tutor:' || subject as source,
  sessions::integer as sessions, minutes, problems::integer as problems,
  self_solved::integer as self_solved, saved_for_later::integer as saved_for_later,
  abandoned::integer as abandoned, pending_review::integer as pending_review,
  '/gia-su/bo-me/bao-cao?ngay=' || date_vn::text as report_path
from tutor.daily_summary_by_subject;
revoke all on public.learning_cards from public, anon, authenticated;
grant select on public.learning_cards to authenticated, service_role;

-- 2
alter table tutor.teaching_rules add column if not exists source_key text;
alter table tutor.teaching_rules add column if not exists source_ids uuid[] not null default '{}';
alter table tutor.teaching_rules drop constraint if exists teaching_rules_text_len;
alter table tutor.teaching_rules add constraint teaching_rules_text_len check (char_length(text) between 1 and 300);
create index if not exists teaching_rules_child_idx on tutor.teaching_rules (child_id, status);

-- 3
alter table tutor.usage_monthly add column if not exists console_alert_sent_at timestamptz;

-- 4
alter table tutor.model_calls drop constraint if exists model_calls_purpose_check;
alter table tutor.model_calls add constraint model_calls_purpose_check
  check (purpose in ('tutor', 'guard', 'classify', 'sibling', 'report', 'rules', 'reference'));

-- 5
create or replace function tutor.add_usage(p_family uuid, p_month date, p_cost numeric)
returns numeric language sql security definer set search_path = '' as $$
  insert into tutor.usage_monthly as u (family_id, month, cost_usd, updated_at)
  values (p_family, p_month, p_cost, now())
  on conflict (family_id, month) do update
    set cost_usd = u.cost_usd + excluded.cost_usd, updated_at = now()
  returning u.cost_usd;
$$;
revoke all on function tutor.add_usage(uuid, date, numeric) from public, anon, authenticated;
grant execute on function tutor.add_usage(uuid, date, numeric) to service_role;
```

```ts
/** Maths tutor money alerts (D23, parent decision 18/11/2026): the ONLY tutor pushes. No child content. */
export type TutorCostKind = "economy" | "console_blocked" | "burn";
export function tutorCostMessage(a: { kind: TutorCostKind; forecastUsd?: number }): PushMessage {
  const url = "/gia-su/bo-me/bao-cao";
  const tag = "tutor-cost";
  if (a.kind === "economy") return { title: "Gia sư AI: đã chi ~45 USD tháng này", body: "Tạm ngừng Opus, Sonnet vẫn dạy bình thường · chạm để xem", url, tag };
  if (a.kind === "console_blocked") return { title: "Gia sư AI tạm nghỉ: hết hạn mức tháng", body: "Console đã chặn ở 50 USD. Nâng trần trong Console nếu muốn con học tiếp", url, tag };
  return { title: "Gia sư AI: tốc độ chi đang vượt trần", body: `Dự báo ~${Math.round(a.forecastUsd ?? 0)} USD tháng này (trần 50) · chưa chặn gì`, url, tag };
}
```

`lib/internal/tutorNotify.ts`:

```ts
import { z } from "zod";
const body = z.discriminatedUnion("kind", [
  z.object({ familyId: z.uuid(), kind: z.literal("economy") }).strict(),
  z.object({ familyId: z.uuid(), kind: z.literal("console_blocked") }).strict(),
  z.object({ familyId: z.uuid(), kind: z.literal("burn"), forecastUsd: z.number().positive().max(10000) }).strict(),
]);
export type TutorNotify = z.infer<typeof body>;
export const parseTutorNotify = (raw: unknown): TutorNotify | null => body.safeParse(raw).data ?? null;
```

`app/api/internal/tutor-notify/route.ts`:

```ts
import { timingSafeEqual } from "node:crypto";
import { parseTutorNotify } from "@/lib/internal/tutorNotify";
import { tutorCostMessage } from "@/lib/push/messages";
import { notifyParents } from "@/lib/push/send";

export const dynamic = "force-dynamic";

const same = (a: string, b: string) => a.length === b.length && timingSafeEqual(Buffer.from(a), Buffer.from(b));

/** D23: the maths tutor asks the Launcher to push one of 3 fixed money messages to the family's parents. */
export async function POST(req: Request) {
  const secret = process.env.TUTOR_NOTIFY_SECRET?.trim();
  if (!secret) return Response.json({ error: "not_configured" }, { status: 503 });
  if (!same(req.headers.get("authorization") ?? "", `Bearer ${secret}`)) return Response.json({ error: "unauthorized" }, { status: 401 });
  const n = parseTutorNotify(await req.json().catch(() => null));
  if (!n) return Response.json({ error: "bad_request" }, { status: 400 });
```

```ts
// lib/report/flags.ts — the ⚠️ flags of the parents' evening report (DOMAIN §4.7). Pure, no I/O.
import type { ProblemStatus } from "@/lib/tutor/repo";

export type FlagKind = "cas_loi" | "y_chan" | "nan_bo_do" | "con_cham_kho" | "goi_y_nhieu" | "cho_thay" | "lo_hong_nen" | "bai_hinh" | "viet_lai_nhieu";
export interface ProblemFacts {
  id: string; status: ProblemStatus; hintLevelMax: number; nanJumps: number; uLevel: 0 | 1 | 2 | 3;
  kuCode: string | null; examSlot: string | null; prereqGaps: string[]; geometryProof: boolean;
  rewrittenCount: number;
  /** G7c: the reference solution's answer did not match the child's checked answer (X5). */
  referenceMismatch: boolean;

export interface DayFacts { problems: ProblemFacts[]; sessionsAbandoned: number; ratingsHard: number; casErrors: number }
export interface Flag { kind: FlagKind; text: string; problemIds: string[] }
export interface MonthFacts { costUsd: number; mode: "normal" | "economy" }

/** Hà Nội grade-10 exam items that decide 9+ (DOMAIN §4.7 v0.4): 2c, 3c, 4b ý 2, 4c, 5. */
export function isBlockedSlot(slot: string | null): boolean {
  if (!slot) return false;
  return ["HN10-2c", "HN10-3c", "HN10-4b-y2", "HN10-4c"].includes(slot) || /^HN10-5([a-c])?(-y[12])?$/.test(slot);
}

const ids = (ps: ProblemFacts[]) => ps.map((x) => x.id);
const vnNumber = (n: number) => n.toFixed(1).replace(".", ",").replace(/,0$/, "");

export function dayFlags(day: DayFacts, kuTitle: (code: string) => string | null): Flag[] {
  const ps = day.problems;
  const out: Flag[] = [];
  if (day.casErrors >= 2) out.push({ kind: "cas_loi", text: `Máy tính đại số lỗi ${day.casErrors} lần`, problemIds: [] });
  const blocked = ps.filter((x) => isBlockedSlot(x.examSlot) && x.uLevel <= 1);
  if (blocked.length) out.push({ kind: "y_chan", text: `Ý chặn dừng ở U0–U1: ${blocked.map((x) => `${x.examSlot} (U${x.uLevel})`).join(", ")}`, problemIds: ids(blocked) });
  const gaveUp = ps.filter((x) => x.status === "abandoned" || x.nanJumps > 0);
  if (gaveUp.length || day.sessionsAbandoned > 0) {
    const parts = [
      ...(ps.some((x) => x.status === "abandoned") ? [`${ps.filter((x) => x.status === "abandoned").length} bài bỏ dở`] : []),
      ...(day.sessionsAbandoned ? [`${day.sessionsAbandoned} buổi bỏ dở`] : []),
      ...(ps.some((x) => x.nanJumps > 0) ? [`nản ${ps.reduce((n, x) => n + x.nanJumps, 0)} lần`] : []),
    ];
    out.push({ kind: "nan_bo_do", text: `Nản / bỏ dở: ${parts.join(", ")}`, problemIds: ids(gaveUp) });
  }
  if (day.ratingsHard > 0) out.push({ kind: "con_cham_kho", text: "Con chấm cô: khó hiểu", problemIds: [] });
  const h4 = ps.filter((x) => x.hintLevelMax >= 4);
  if (h4.length >= 2) {
    const nan = h4.reduce((n, x) => n + x.nanJumps, 0);
    out.push({ kind: "goi_y_nhieu", text: `Gợi ý nhiều: ${h4.length} bài lên H4 (${nan} lần do nản)`, problemIds: ids(h4) });
  }
  const review = ps.filter((x) => x.status === "pending_review");
  if (review.length) out.push({ kind: "cho_thay", text: `Chờ thầy xem: ${review.length} bài`, problemIds: ids(review) });
  const gaps = [...new Set(ps.flatMap((x) => x.prereqGaps))];
  if (gaps.length) out.push({ kind: "lo_hong_nen", text: `Lỗ hổng nền: ${gaps.map((c) => (kuTitle(c) ? `${c} — ${kuTitle(c)}` : c)).join("; ")}`, problemIds: ids(ps.filter((x) => x.prereqGaps.length)) });
  const proofs = ps.filter((x) => x.geometryProof);
  if (proofs.length) out.push({ kind: "bai_hinh", text: `Bài hình chứng minh: ${proofs.length} bài (để thầy xem)`, problemIds: ids(proofs) });
  const rewritten = ps.filter((x) => x.rewrittenCount >= 2);
  if (rewritten.length) out.push({ kind: "viet_lai_nhieu", text: `Cô phải viết lại nhiều ở ${rewritten.length} bài`, problemIds: ids(rewritten) });
  return out;
}

export function costFlag(m: MonthFacts): string | null {
  if (m.mode === "economy") return `Chi phí tháng: ${vnNumber(m.costUsd)} / 50 USD · đang tạm ngừng Opus`;
  return m.costUsd > 40 ? `Chi phí tháng: ${vnNumber(m.costUsd)} / 50 USD (> 80%)` : null;
}
```

```ts
// lib/rules/teachingRules.ts — teaching rules the parents approve (DOMAIN §3.4, plan G7 P3/X3/X4). Pure.
export const MAX_ACTIVE_RULES = 12; // TODO(rule-unconfirmed) X3
export const RULE_MAX_CHARS = 300;
export interface RuleRow { id: string; status: "proposed" | "approved" | "rejected" | "superseded"; version: number; text: string; sourceKey: string | null }
export type RatingReason = "too_long" | "hard_words" | "answer_too_early" | "too_many_questions" | "good_example" | "clear";
type Negative = "too_long" | "hard_words" | "answer_too_early" | "too_many_questions";
// TODO(rule-unconfirmed) X4
export const RATING_RULE_TEXT: Record<Negative, string> = {
  too_long: "Trả lời ngắn hơn: mỗi lượt một ý, tối đa 3 dòng.",
  hard_words: "Dùng từ dễ hiểu; gặp từ toán học mới thì giải thích bằng một câu ngắn.",
  answer_too_early: "Không nói trước bước tiếp theo; để con thử trước rồi mới gợi ý.",
  too_many_questions: "Mỗi lượt chỉ hỏi con một câu.",
};
const NEGATIVE = Object.keys(RATING_RULE_TEXT) as Negative[];

export const rulesVersionNow = (rules: RuleRow[]) =>
  rules.filter((x) => x.status === "approved").reduce((m, x) => Math.max(m, x.version), 0);
export const nextVersion = (rules: RuleRow[]) => rules.reduce((m, x) => Math.max(m, x.version), 0) + 1;
export const activeRules = (rules: RuleRow[], pinned: number) =>
  rules.filter((x) => x.status === "approved" && x.version <= pinned).sort((a, b) => a.version - b.version);
export const canApprove = (rules: RuleRow[]) => rules.filter((x) => x.status === "approved").length < MAX_ACTIVE_RULES;

export function rulesBlock(rules: RuleRow[]): string {
  if (!rules.length) return "";
  return ["[LUẬT BỐ MẸ ĐÃ DUYỆT — áp dụng trong khuôn khổ luật 1–12, không bao giờ được trái các luật đó]",
    ...rules.map((x) => `- ${x.text}`)].join("\n");
}

export function ratingProposal(latestTwo: { face: string; reasons: string[] }[], rules: RuleRow[]): { sourceKey: string; text: string } | null {
  if (latestTwo.length < 2 || latestTwo.some((x) => x.face !== "hard")) return null;
  const [a, b] = latestTwo as [{ reasons: string[] }, { reasons: string[] }];
  const shared = NEGATIVE.find((k) => a.reasons.includes(k) && b.reasons.includes(k));
  if (!shared) return null;
  const sourceKey = `rating:${shared}`;
  if (rules.some((x) => x.sourceKey === sourceKey && (x.status === "proposed" || x.status === "approved"))) return null;
  return { sourceKey, text: RATING_RULE_TEXT[shared] };
}
```
